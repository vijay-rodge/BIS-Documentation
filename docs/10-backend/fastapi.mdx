# FastAPI Application & Routers

The application entrypoint initializes middleware, registers routers, and manages the database connection lifecycle.

```python
# backend/main.py
from contextlib import asynccontextmanager
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from backend.api.routes import router as api_router
from backend.config.settings import settings

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Startup: Initialize DB connection pool & verify seed data
    logger.info("Initializing BIS Intelligent Assistant Backend...")
    yield
    # Shutdown: Close database pools and external connections
    logger.info("Shutting down BIS Intelligent Assistant Backend...")

app = FastAPI(
    title=settings.PROJECT_NAME,
    version="1.0.0",
    description="Authoritative AI Assistant for Indian Standards & BIS Services",
    lifespan=lifespan
)

# CORS Security Middleware
app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.CORS_ORIGINS,
    allow_credentials=True,
    allow_methods=["GET", "POST", "OPTIONS"],
    allow_headers=["*"]
)

# Register modular API router
app.include_router(api_router, prefix="/api/v1")
```
