API Integration Testing

Integration tests verify that HTTP endpoints correctly interact with database sessions and return conformant JSON schemas.

# tests/test_api.py
import pytest
from fastapi.testclient import TestClient
from backend.main import app

client = TestClient(app)

def test_health_check_endpoint():
    response = client.get("/api/v1/health")
    assert response.status_code == 200
    data = response.json()
    assert data["status"] == "healthy"
    assert "database" in data

def test_chat_completions_endpoint():
    payload = {"query": "What standard applies to motorcycle helmets?"}
    response = client.post("/api/v1/chat/completions", json=payload)
    assert response.status_code == 200
    data = response.json()
    assert data["confidence"] == "HIGH"
    assert "IS 4151" in data["answer"]
    assert len(data["sources"]) > 0
